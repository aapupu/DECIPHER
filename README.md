# DECIPHER

**DECoupled Invariant Prototype-based deconvolution for Heterogeneous
Expression Reconstruction**

DECIPHER estimates cell-type proportions from bulk or spatial mixtures using
single-cell-derived pseudo mixtures, unlabeled real mixtures, and cell-type
prototypes.

## Installation

```bash
git clone https://github.com/aapupu/DECIPHER.git
cd DECIPHER
conda env create -f environment.yml
conda activate decipher
```

GPU is used automatically when CUDA is available.

## Input data

Single-cell input is AnnData (`.h5ad`) with `obs["celltype"]` and
`obsm["latent_gex"]`. Mixture data (pseudo / real) is generated by the
program from single-cell data via `DecodePseudoBulkBuilder`.

## Usage

### 1. sc_data

```python
import numpy as np
import scanpy as sc
import torch
from sklearn.model_selection import train_test_split
from torch.utils.data import DataLoader

from DECIPHER.dataprocess import (
    DecodePseudoBulkBuilder,
    DecodePseudoBulkConfig,
    process_adata,
    reorder_celltype_proportions,
    row_max_normalize,
)
from DECIPHER.dataset import PseudoRealDynamicPairDataset, pair_collate
from DECIPHER.loss import LossWeights
from DECIPHER.model import DECIPHER
from DECIPHER.utils import seed_everything

seed_everything(42)
device = "cuda" if torch.cuda.is_available() else "cpu"

adata = sc.read_h5ad("sc_reference.h5ad")
celltype_order = list(dict.fromkeys(adata.obs["celltype"].astype(str)))

sc_data = process_adata(
    adata,
    celltype_key="celltype",
    state_key=None,
    assume_log1p_cp10k=False,  # set True if already log1p-normalized
)

latent = np.asarray(adata.obsm["latent_gex"], dtype=np.float32)
labels = adata.obs["celltype"].astype(str).to_numpy()
s_z = torch.from_numpy(
    np.stack([latent[labels == ct].mean(0) for ct in celltype_order]).astype(np.float32)
)
```

### 2. Mix data

```python
N_PSEUDO, N_REAL, CELLS_PER_MIXTURE = 8000, 1000, 200

# pseudo mixtures from the reference
builder = DecodePseudoBulkBuilder(
    sc_data,
    DecodePseudoBulkConfig(cells_per_bulk=CELLS_PER_MIXTURE, seed=42),
    require_state=False,
)
pseudo_x, pseudo_prop = [], []
for _ in range(N_PSEUDO):
    x, p, _ = builder.build_one()
    pseudo_x.append(x)
    pseudo_prop.append(reorder_celltype_proportions(p, sc_data.celltype_map, celltype_order))
pseudo_x = row_max_normalize(np.stack(pseudo_x))
pseudo_prop = np.stack(pseudo_prop).astype(np.float32)

# real mixtures synthesized from held-out single-cell data
adata_real = sc.read_h5ad("sc_real.h5ad")
sc_real = process_adata(
    adata_real,
    celltype_key="celltype",
    state_key=None,
    assume_log1p_cp10k=False,
)
real_builder = DecodePseudoBulkBuilder(
    sc_real,
    DecodePseudoBulkConfig(cells_per_bulk=CELLS_PER_MIXTURE, seed=2024),
    require_state=False,
)
real_x, real_prop = [], []
for _ in range(N_REAL):
    x, p, _ = real_builder.build_one()
    real_x.append(x)
    real_prop.append(reorder_celltype_proportions(p, sc_real.celltype_map, celltype_order))
real_x = row_max_normalize(np.stack(real_x))
real_prop = np.stack(real_prop).astype(np.float32)
real_domain = np.ones(real_x.shape[0], dtype=np.int64)

idx_train, idx_val = train_test_split(np.arange(N_PSEUDO), test_size=0.2, random_state=42)
train_dataset = PseudoRealDynamicPairDataset(
    pseudo_x[idx_train], pseudo_prop[idx_train], real_x, real_domain, seed=42
)
val_dataset = PseudoRealDynamicPairDataset(
    pseudo_x[idx_val], pseudo_prop[idx_val], real_x, real_domain, seed=43
)
```

### 3. Train model

```python
train_loader = DataLoader(train_dataset, batch_size=128, shuffle=True, collate_fn=pair_collate)
val_loader = DataLoader(val_dataset, batch_size=128, shuffle=False, collate_fn=pair_collate)

model = DECIPHER(
    s_z=s_z,
    n_feature=pseudo_x.shape[1],
    hidden_dim=(128, 128),
    n_domain=2,
).to(device)

model.fit(
    train_loader,
    val_dataloader=val_loader,
    lr=1e-4,
    weight_decay=1e-3,
    max_epoch=500,
    device=device,
    patience=20,
    loss_weight=LossWeights(
        w_prop=100, w_rec=0.1, w_latrec=0.1, w_dom=0.2, w_align=0.3, w_contrast=0.1, mse_ce_ratio=20
    ),
    outdir="DECIPHER.pt",
    verbose=True,
)
```

### 4. Model Prediction

```python
model.load_state_dict(torch.load("DECIPHER.pt", map_location=device))
model.to(device)

x = torch.from_numpy(real_x).to(device)
_, proportions = model.deconvolution(x)
proportions = proportions.cpu().numpy()  # columns follow celltype_order

z_c, z_s = model.encode_z(x)
```
